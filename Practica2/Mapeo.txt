=================================
SW-CORE-A
=================================
enable
configure terminal
hostname SW-CORE-A
no ip domain-lookup
enable secret 202201336
service password-encryption
banner motd ^CACCESO SOLO AUTORIZADO - PRACTICA2^C

spanning-tree mode rapid-pvst

vtp domain 202201336
vtp mode server
vtp password area1
vtp version 2

vlan 16
 name QUIROFANO
vlan 26
 name UCI
vlan 36
 name ADMIN
vlan 46
 name FARMACIA
vlan 56
 name LAB
vlan 99
 name NATIVE
vlan 999
 name BLACKHOLE

interface range fa0/7-9
 description ETH-CH-TO-SW-CORE-B
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 16,26,36,46,56,99,999
 channel-group 1 mode desirable
 no shutdown
exit

interface port-channel 1
 description PO1-COREA-COREB
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 16,26,36,46,56,99,999
 no shutdown
exit

interface range fa0/1-6
 description TRUNK-TO-HOSPITAL-SW
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 16,26,36,46,56,99,999
 no shutdown
exit

interface range fa0/10-24
 description UNUSED-BLACKHOLE
 switchport mode access
 switchport access vlan 999
 shutdown
exit

spanning-tree vlan 16,26,36,46,56,99 root primary

line console 0
 password 202201336
 login
line vty 0 15
 password 202201336
 login
end
write memory
